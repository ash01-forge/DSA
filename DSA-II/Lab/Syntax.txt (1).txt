#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

bool first_asc_second_asc(pair<int,int>a, pair<int,int>b)
{
    if(a.first!=b.first)
        return a.first<b.first;
    return a.second<b.second;
}

bool first_asc_second_desc(pair<int,int>a, pair<int,int>b)
{
    if(a.first!=b.first)
        return a.first<b.first;
    return a.second>b.second;
}

bool second_ascending(pair<int,int>a, pair<int,int>b)
{
    if(a.second!=b.second)
        return a.second<b.second;
    return a.first<b.first;
}

bool sum_ascending(pair<int,int>a, pair<int,int>b)
{
    return a.first+a.second < b.first+b.second;
}

bool sum_descending(pair<int,int>a, pair<int,int>b)
{
    return a.first+a.second > b.first+b.second;
}

int main()
{
    int n;
    cin>>n;

    vector<pair<int,int>>v;
    for(int i=0;i<n;i++)
    {
        int a,b;
        cin>>a>>b;
        v.push_back(make_pair(a,b));
    }

    int choice;
    cin>>choice;

    if(choice==1)
        sort(v.begin(),v.end(),first_asc_second_asc);
    else if(choice==2)
        sort(v.begin(),v.end(),first_asc_second_desc);
    else if(choice==3)
        sort(v.begin(),v.end(),second_ascending);
    else if(choice==4)
        sort(v.begin(),v.end(),sum_ascending);
    else
        sort(v.begin(),v.end(),sum_descending);

    for(int i=0;i<v.size();i++)
        cout<<v[i].first<<" "<<v[i].second<<endl;
}

[DNC-01] Merge Sort in Descending Order
----------------------------------------------------------------------------------------
#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int>v;

void _merge_sort_descending(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(_end-start==1) /// size 2
    {
        if(v[start]<v[_end])
        {
            swap(v[start],v[_end]);
        }
        return;
    }
    else if(start==_end) /// size 1
        return;

    ///step-1: divide
    int mid=(start+_end)/2;
    _merge_sort_descending(start,mid); /// sort from start to mid
    _merge_sort_descending(mid+1,_end);/// sort from mid+1 to end


    ///step-3: merge
    vector<int>l,r; /// l=v[start...mid], r=v[mid+1..._end]
    for(int i=start;i<=mid;i++)
        l.push_back(v[i]);
    for(int i=mid+1;i<=_end;i++)
        r.push_back(v[i]);

    int i=0,j=0;
    vector<int>new_array;

    while(i<l.size() && j<r.size())
    {
        /// larger element will come first
        if(l[i]>r[j])
        {
            new_array.push_back(l[i]);
            i++;
        }
        else
        {
            new_array.push_back(r[j]);
            j++;
        }
    }

    while(i<l.size())
    {
        new_array.push_back(l[i]);
        i++;
    }

    while(j<r.size())
    {
        new_array.push_back(r[j]);
        j++;
    }

    /// copy new_array in our main vector v
    int p=0;
    for(int k=start;k<=_end;k++)
    {
        v[k]=new_array[p];
        p++;
    }
}

int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        int temp;
        cin>>temp;
        v.push_back(temp);
    }

    if(n>0)
        _merge_sort_descending(0,n-1);

    cout<<"after sorting in descending order-----------"<<endl;
    for(int i=0;i<n;i++)
        cout<<v[i]<<" ";
}

/*

-5 -10 16 2 6 5 3 2 19
*/

/// sample output
/*

[DNC-02] Three-Way Merge Sort
----------------------------------------------------------------------------------------
#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int>v;

void _merge_sort_3_way(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(_end-start==1) /// size 2
    {
        if(v[start]>v[_end])
        {
            swap(v[start],v[_end]);
        }
        return;
    }
    else if(start==_end) /// size 1
        return;

    ///step-1: divide the array into 3 sections
    int mid1=start+(_end-start)/3;
    int mid2=start+2*(_end-start)/3;

    _merge_sort_3_way(start,mid1);
    _merge_sort_3_way(mid1+1,mid2);
    _merge_sort_3_way(mid2+1,_end);


    ///step-3: merge 3 sorted arrays
    vector<int>l,m,r;

    for(int i=start;i<=mid1;i++)
        l.push_back(v[i]);
    for(int i=mid1+1;i<=mid2;i++)
        m.push_back(v[i]);
    for(int i=mid2+1;i<=_end;i++)
        r.push_back(v[i]);

    int i=0,j=0,k=0;
    vector<int>new_array;

    /// all three pointers are inside the bounds
    while(i<l.size() && j<m.size() && k<r.size())
    {
        if(l[i]<=m[j] && l[i]<=r[k])
        {
            new_array.push_back(l[i]);
            i++;
        }
        else if(m[j]<=l[i] && m[j]<=r[k])
        {
            new_array.push_back(m[j]);
            j++;
        }
        else
        {
            new_array.push_back(r[k]);
            k++;
        }
    }

    /// merge the two remaining arrays
    while(i<l.size() && j<m.size())
    {
        if(l[i]<=m[j]) new_array.push_back(l[i++]);
        else new_array.push_back(m[j++]);
    }

    while(i<l.size() && k<r.size())
    {
        if(l[i]<=r[k]) new_array.push_back(l[i++]);
        else new_array.push_back(r[k++]);
    }

    while(j<m.size() && k<r.size())
    {
        if(m[j]<=r[k]) new_array.push_back(m[j++]);
        else new_array.push_back(r[k++]);
    }

    /// copy the remaining elements
    while(i<l.size()) new_array.push_back(l[i++]);
    while(j<m.size()) new_array.push_back(m[j++]);
    while(k<r.size()) new_array.push_back(r[k++]);

    /// copy new_array in our main vector v
    int p=0;
    for(int index=start;index<=_end;index++)
    {
        v[index]=new_array[p];
        p++;
    }
}

int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        int temp;
        cin>>temp;
        v.push_back(temp);
    }

    if(n>0)
        _merge_sort_3_way(0,n-1);

    cout<<"after 3-way merge sorting-----------"<<endl;
    for(int i=0;i<n;i++)
        cout<<v[i]<<" ";
}

15 2 19 -5 7 3 3 20 1 8
*/

#include<iostream>

int n;
vector<int>v;

int even_sum(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(start==_end) /// size 1
    {
        if(v[start]%2==0)
            return v[start];
        else
            return 0;
    }

    ///step-1: divide
    int mid=(start+_end)/2;
    int left_even_sum=even_sum(start,mid);
    int right_even_sum=even_sum(mid+1,_end);


    ///step-3: merge
    int merged_even_sum=left_even_sum+right_even_sum;
    return merged_even_sum;
}

#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int>v;

/// the pair will store (maximum, second maximum)
pair<int,int> find_second_maximum(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(_end-start==1) /// size 2
    {
        if(v[start]>v[_end])
            return make_pair(v[start],v[_end]);
        else
            return make_pair(v[_end],v[start]);
    }
    else if(start==_end) /// size 1
    {
        return make_pair(v[start],-100000000);
    }

    ///step-1: divide
    int mid=(start+_end)/2;
    pair<int,int>left_answer=find_second_maximum(start,mid);
    pair<int,int>right_answer=find_second_maximum(mid+1,_end);


    ///step-3: merge
    /// candidates are max1, second_max1, max2, second_max2
    vector<int>candidate;
    candidate.push_back(left_answer.first);
    candidate.push_back(left_answer.second);
    candidate.push_back(right_answer.first);
    candidate.push_back(right_answer.second);

    int maximum=-100000000;
    int second_maximum=-100000000;

    for(int i=0;i<candidate.size();i++)
    {
        if(candidate[i]>maximum)
        {
            second_maximum=maximum;
            maximum=candidate[i];
        }
        else if(candidate[i]>second_maximum)
        {
            second_maximum=candidate[i];
        }
    }

    return make_pair(maximum,second_maximum);
}

int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        int temp;
        cin>>temp;
        v.push_back(temp);
    }

    if(n>=2)
    {
        pair<int,int>answer=find_second_maximum(0,n-1);
        cout<<"maximum = "<<answer.first<<endl;
        cout<<"second maximum = "<<answer.second<<endl;
    }
}

[DNC-06] Calculate b^n Using Fast Power
----------------------------------------------------------------------------------------
#include<iostream>
using namespace std;
long long b;
int n;

long long calculate_power(int exponent)
{
    ///step-2: conquer (base condition)
    if(exponent==0)
        return 1;
    else if(exponent==1)
        return b;

    ///step-1: divide
    int half_exponent=exponent/2;
    long long half_answer=calculate_power(half_exponent);


    ///step-3: merge
    long long answer=half_answer*half_answer;

    /// if exponent is odd, multiply one extra b
    if(exponent%2==1)
        answer=answer*b;

    return answer;
}

int main()
{
    cin>>b>>n;

[DNC-07] Count Multiples of 3 or 5
----------------------------------------------------------------------------------------
#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int>v;

int count_multiple_3_or_5(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(start==_end) /// size 1
    {
        if(v[start]%3==0 || v[start]%5==0)
            return 1;
        else
            return 0;
    }

    ///step-1: divide
    int mid=(start+_end)/2;
    int left_count=count_multiple_3_or_5(start,mid);
    int right_count=count_multiple_3_or_5(mid+1,_end);


    ///step-3: merge
    int merged_count=left_count+right_count;
    return merged_count;
}

int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        int temp;
        cin>>temp;
        v.push_back(temp);
    }

    if(n>0)
        cout<<count_multiple_3_or_5(0,n-1)<<endl;
}

[DNC-08] Minimum-Sum Subarray
----------------------------------------------------------------------------------------
#include<iostream>
#include<vector>
using namespace std;
int n;
vector<int>v;

int minsum_subarray(int start, int _end)
{
    ///step-2: conquer (base condition)
    if(start==_end) /// size 1
        return v[start];


    ///step-1: divide
    int mid=(start+_end)/2;
    int leftmin=minsum_subarray(start,mid);
    int rightmin=minsum_subarray(mid+1,_end);


    ///step-3: merge

    int lm=100000000; /// from mid -> start, tracker of minimum sum
    int current_sum=0;

    for(int i=mid;i>=start;i--)
    {
        current_sum+=v[i];
        lm=min(lm,current_sum);
    }

    int rm=100000000; /// from mid+1 -> end, tracker of minimum sum
    current_sum=0;

    for(int i=mid+1;i<=_end;i++)
    {
        current_sum+=v[i];
        rm=min(rm,current_sum);
    }

    int merged_min=lm+rm;
    int ans=min(merged_min,min(leftmin,rightmin));

    return ans;
}

int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        int temp;
        cin>>temp;
        v.push_back(temp);
    }

    if(n>0)
    {
        int minsum=minsum_subarray(0,n-1);
        cout<<minsum<<endl;
    }
}
